Private Sub Form_Load()
 DoCmd.MoveSize , , 17010, 11300
  Dim strCourseNum As String
    
    ' باز کردن پنجره براي پرسيدن شماره دوره
    strCourseNum = InputBox("لطفاً شماره دوره مورد نظر را وارد کنيد:" & vbCrLf & _
    
    ' بررسي اينکه آيا کاربر شماره‌اي وارد کرده است يا خير
    If strCourseNum <> "" Then
        ' اعمال فيلتر بر اساس نام فيلدهاي دقيق جدول شما
        Me.Filter = "studentID IN (SELECT studentRef FROM TblEnroll WHERE courseNumber = " & strCourseNum & ")"
        Me.FilterOn = True
    Else
        ' اگر کادر خالي بود يا Cancel زده شد، فيلتر خاموش بماند تا همه ليست ديده شود
        Me.FilterOn = False
    End If
End Sub
