Private Sub btnClearFilter_Click()

    Me.todate = Null
    
    ' ?. برداشتن فيلتر

    Me.OrderBy = "ExpenseDate DESC"
    Me.OrderByOn = True
    
    End Sub

Private Sub btnSearch_Click()

    On Error Resume Next
    
    Dim strFilter As String
    
    ' ?. چک کردن اينکه هر دو تاريخ پر شده باشند
    If IsNull(Me.fromex) Or IsNull(Me.todate) Then
        MsgBox "لطفاً هر دو تاريخ (از - تا) را وارد کنيد.", vbExclamation, "خطا در جستجو"
        Exit Sub
    End If
    
    ' ?. ساختن شرط فيلتر (بازه زماني)
    ' فرض کرديم اسم فيلد تاريخ در جدول شما ExpenseDate است. (اگر چيز ديگري است، عوض کنيد)
    ' چون تاريخ شمسي متني است، از علامت تک‌کوتيشن ' استفاده مي‌کنيم
    strFilter = "ExpenseDate >= '" & Me.fromex & "' AND ExpenseDate <= '" & Me.todate & "'"
    
    ' ?. اعمال فيلتر
    Me.Filter = strFilter
    Me.FilterOn = True
    
    ' ?. مرتب‌سازي (اختياري: جديدترين‌ها بالا باشند يا قديمي‌ترين‌ها؟)
    Me.OrderBy = "ExpenseDate DESC"
    Me.OrderByOn = True
    
End Sub
