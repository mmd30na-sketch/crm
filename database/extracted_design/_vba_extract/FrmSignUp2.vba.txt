Private Sub Form_BeforeUpdate(Cancel As Integer)
    Dim strMissing As String
    strMissing = ""
    
    ' بررسي فيلدها
    If IsNull(Me.firstName) Or Me.firstName = "" Then strMissing = strMissing & "- نام" & vbCrLf
    If IsNull(Me.lastName) Or Me.lastName = "" Then strMissing = strMissing & "- نام خانوادگي" & vbCrLf
    If IsNull(Me.Ncode) Or Me.Ncode = "" Then strMissing = strMissing & "- کدملي" & vbCrLf
    If IsNull(Me.phoneNumber) Or Me.phoneNumber = "" Then strMissing = strMissing & "-شماره تماس" & vbCrLf
    ' اگر ناقص بود
    If strMissing <> "" Then
        If MsgBox("فيلدهاي زير خالي هستند:" & vbCrLf & strMissing & vbCrLf & _
                  "آيا ذخيره انجام شود؟ (اگر No بزنيد ذخيره نمي‌شود)", _
                  vbQuestion + vbYesNo + vbDefaultButton2) = vbNo Then
            
            Cancel = True ' اين دستور خيلي مهم است: جلوي ذخيره را مي‌گيرد
            ' Me.Undo ' (اختياري) اگر بخواهي تغييرات تايپ شده هم پاک شود اين را فعال کن
           MsgBox "رکورد ذخيره نشد ، اطلاعات را کامل کنيد"
        End If
    End If
End Sub

Private Sub scnIdcartBtn_Click()
    On Error GoTo ErrorHandler

    ' 1. Check for Last Name and ID
    If IsNull(Me.lastName) Or IsNull(Me.StudentID) Then
        MsgBox "ابتدا نام و نام خانوادگي را وارد کنيد", vbExclamation
        Exit Sub
    End If
        
            If IsNull(Me.TblEnrollsubform.Form!courseNumber) Or Me.TblEnrollsubform.Form!courseNumber = "" Then
        MsgBox "لطفاً ابتدا شماره دوره را انتخاب کنيد!", vbExclamation
        Exit Sub
    End If
            Dim strCourseFolder As String
    Dim Manager As Object
    Dim ImageFile As Object
    Dim Dialog As Object
    Dim IP As Object
    Dim strMainFolder As String
    Dim strUserFolder As String
    Dim strFilePath As String
    Dim rsParent As DAO.Recordset
    Dim rsChild As DAO.Recordset2
    
    ' Variables for Cropping
    Dim TotalWidth As Long
    Dim TotalHeight As Long
    Dim TargetWidth As Long
    Dim TargetHeight As Long
    
    ' --- SETTINGS: TARGET SIZE ---
    TargetWidth = 1100
    TargetHeight = 750
    ' -----------------------------

    
    ' 2. Create Folders
    ' الف) پوشه اصلي
    strMainFolder = CurrentProject.Path & "\StudentFiles\"
    If Dir(strMainFolder, vbDirectory) = "" Then MkDir strMainFolder
    
    ' ب) پوشه شماره دوره
    strCourseFolder = strMainFolder & Me.TblEnrollsubform.Form!courseNumber & "\"
    If Dir(Left(strCourseFolder, Len(strCourseFolder) - 1), vbDirectory) = "" Then
        MkDir Left(strCourseFolder, Len(strCourseFolder) - 1)
    End If
    
    ' ج) پوشه کارآموز بر اساس نام و کد
    strUserFolder = strCourseFolder & Me.lastName & "_" & Me.StudentID & "\"
    If Dir(Left(strUserFolder, Len(strUserFolder) - 1), vbDirectory) = "" Then
        MkDir Left(strUserFolder, Len(strUserFolder) - 1)
    End If

    ' 3. Setup Scanner
    Set Manager = CreateObject("WIA.DeviceManager")
    If Manager.DeviceInfos.Count = 0 Then
        MsgBox "اسکنر يافت نشد", vbCritical
        Exit Sub
    End If
    
    ' 4. Scan
    Set Dialog = CreateObject("WIA.CommonDialog")
    Set ImageFile = Dialog.ShowAcquireImage(1, 1, 65536, "{B96B3CAE-0728-11D3-9D7B-0000F81EF32E}", False, True, False)
    
    If ImageFile Is Nothing Then Exit Sub
    
    ' 5. CROP LOGIC
    Set IP = CreateObject("WIA.ImageProcess")
    IP.Filters.Add IP.FilterInfos("Crop").FilterID
    
    TotalWidth = ImageFile.Width
    TotalHeight = ImageFile.Height
    
    If TotalWidth > TargetWidth Then
        IP.Filters(1).Properties("Right") = TotalWidth - TargetWidth
    End If
    
    If TotalHeight > TargetHeight Then
        IP.Filters(1).Properties("Bottom") = TotalHeight - TargetHeight
    End If
    
    IP.Filters(1).Properties("Left") = 0
    IP.Filters(1).Properties("Top") = 0
    
    Set ImageFile = IP.Apply(ImageFile)
    
    ' 6. Save File (THE FIX IS HERE)
    ' Now strUserFolder already has "\" at the end
    strFilePath = strUserFolder & Me.lastName & "_" & Me.StudentID & "_ID.jpg"
    
    If Dir(strFilePath) <> "" Then Kill strFilePath
    ImageFile.SaveFile strFilePath
    
    ' 7. Save to Database
    If Me.Dirty Then Me.Dirty = False

    Set rsParent = Me.RecordsetClone
    rsParent.Bookmark = Me.Bookmark
    rsParent.Edit
    
    Set rsChild = rsParent.Fields("IdCartPht").Value
    
    Do While Not rsChild.EOF
        rsChild.Delete
        rsChild.MoveNext
    Loop
    
    rsChild.AddNew
    rsChild.Fields("FileData").LoadFromFile strFilePath
    rsChild.Update
    
    rsChild.Close
    rsParent.Update
    
    ' 8. Finish
    Me.Refresh
    MsgBox "عکس ذخيره شد " & strUserFolder, vbInformation

    Exit Sub

ErrorHandler:
    MsgBox "خطا " & Err.description
End Sub
